### Install all the conda environments required to run the metqc snakemake pipeline.

## Install the fastqc conda environment.

# Create the fastqc conda environment.
conda create --name fastqc_env

# Activate the fastqc conda environment.
conda activate fastqc_env

# Install the fastqc conda package.
conda install -c conda-forge -c bioconda fastqc=0.11.8 -y


## Install the multiqc conda environment.

# Create the multiqc conda environment.
conda create --name multiqc_env

# Activate the multiqc conda environment.
conda activate multiqc_env

# Install the multiqc conda package.
conda install -c conda-forge -c bioconda multiqc=1.35 -y
