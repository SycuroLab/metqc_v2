# Install the fastqc conda environment.
conda create --name fastqc_env
conda activate fastqc_env
conda install -c conda-forge -c bioconda  fastqc=0.11.8 -y

# Install the multiqc conda environment.
conda create --name multiqc_env
conda activate multiqc_env
conda install -c conda-forge -c bioconda multiqc=1.35 -y

# # Install the prinseq conda environment.

# Install the bmtagger conda environment.

# Install the bbmap conda environment.

# Install the python conda environment.
conda create --name python_env

conda install -c conda-forge python=3.14 -y 
conda install -c conda-forge matplotlib=3.11.2 -y 
conda install -c conda-forge pandas=3.0.6 -y
conda install -c conda-forge numpy=2.5.3 -y

# Install the seqkit conda environment.
conda create --name seqkit_env

# Export conda environments to a yaml file.
conda activate fastqc_env
